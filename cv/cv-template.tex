\documentclass[11pt]{article}

% ----- page: single column, white background, black text, normal margins -----
\usepackage[margin=1in]{geometry}

% ----- fonts -----

\usepackage{xeCJK}

\setCJKmainfont{Noto Sans SC Light}[
  BoldFont = Noto Sans SC,
